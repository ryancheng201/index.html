<script>
  let items = [];

  function toggleInstallSub() {
    const main = document.getElementById('installMain').value;
    document.getElementById('plywoodBox').style.display = (main === 'plywood') ? 'block' : 'none';
    document.getElementById('raisedBox').style.display = (main === 'raised') ? 'block' : 'none';
    runCalc();
  }

  function getSelectedDetails() {
    const wood = document.getElementById('woodType');
    const thick = document.getElementById('thickness');
    const width = document.getElementById('width');
    const len = document.getElementById('length');
    const surf = document.getElementById('surface');
    const paint = document.getElementById('paint');
    const installMain = document.getElementById('installMain');

    const isHerringbone = width.selectedOptions[0].getAttribute('data-custom') === 'true';

    let installCost = 0;
    let installDesc = installMain.selectedOptions[0].getAttribute('data-name');
    let isTatami = false;

    if (installMain.value === 'sealing') {
      installCost = 1200;
    } else if (installMain.value === 'plywood') {
      const ply = document.getElementById('plywoodThick');
      installCost = Number(ply.value);
      installDesc = `夾面式(${ply.selectedOptions[0].getAttribute('data-name')})`;
    } else if (installMain.value === 'raised') {
      const rsd = document.getElementById('raisedType');
      installCost = Number(rsd.value);
      installDesc = `架高式(${rsd.selectedOptions[0].getAttribute('data-name')})`;
      isTatami = rsd.selectedOptions[0].getAttribute('data-tatami') === 'true';
    }

    const unitPrice = Number(wood.value) + Number(thick.value) + Number(width.value) +
                      Number(len.value) + Number(surf.value) + Number(paint.value) + installCost;

    const area = parseFloat(document.getElementById('area').value) || 0;
    const subtotal = Math.round(unitPrice * area);

    const specSummary = `${wood.selectedOptions[0].getAttribute('data-name')} / ` +
                        `${thick.selectedOptions[0].getAttribute('data-name')} / ` +
                        `${width.selectedOptions[0].getAttribute('data-name')} / ` +
                        `${len.selectedOptions[0].getAttribute('data-name')} / ` +
                        `${surf.selectedOptions[0].getAttribute('data-name')} / ` +
                        `${paint.selectedOptions[0].getAttribute('data-name')} / ` +
                        `${installDesc}`;

    return {
      areaName: document.getElementById('areaName').value.trim() || '未命名空間',
      unitPrice,
      area,
      subtotal,
      specSummary,
      isHerringbone,
      isTatami
    };
  }

  function runCalc() {
    const data = getSelectedDetails();

    document.getElementById('liveUnit').textContent = '$' + data.unitPrice.toLocaleString('zh-TW');
    document.getElementById('liveTotal').textContent = '$' + data.subtotal.toLocaleString('zh-TW');

    const notices = [];
    if (data.isHerringbone) notices.push('人字型尺寸／價格另訂，目前未計入寬度加價。');
    if (data.isTatami) notices.push('合室架高以 3,500 元起算，實際價格另議。');
    notices.push('顏色不限，需製作樣品確認。');

    const listEl = document.getElementById('noticeList');
    listEl.innerHTML = notices.map(n => `<li>${n}</li>`).join('');
  }

  function addItemToList() {
    const data = getSelectedDetails();
    if (data.area <= 0) {
      alert('請輸入有效的施作坪數');
      return;
    }
    items.push(data);
    renderTable();
  }

  function removeItem(index) {
    items.splice(index, 1);
    renderTable();
  }

  function clearAllItems() {
    if (items.length === 0) return;
    if (confirm('確定清空所有報價項目？')) {
      items = [];
      renderTable();
    }
  }

  function renderTable() {
    const tbody = document.getElementById('quoteBody');
    const summaryBox = document.getElementById('summaryBox');

    if (items.length === 0) {
      tbody.innerHTML = '<tr><td colspan="6" style="text-align: center; color: #999; padding: 24px 0;">尚未加入任何項目，請從左側設定並加入。</td></tr>';
      summaryBox.style.display = 'none';
      return;
    }

    let totalArea = 0;
    let totalPrice = 0;

    tbody.innerHTML = items.map((item, idx) => {
      totalArea += item.area;
      totalPrice += item.subtotal;
      return `
        <tr>
          <td><strong>${item.areaName}</strong></td>
          <td style="font-size:12px; color:#555;">${item.specSummary}</td>
          <td>$${item.unitPrice.toLocaleString('zh-TW')}</td>
          <td>${item.area}</td>
          <td><strong>$${item.subtotal.toLocaleString('zh-TW')}</strong></td>
          <td class="no-print"><button class="btn-del" onclick="removeItem(${idx})">刪除</button></td>
        </tr>
      `;
    }).join('');

    document.getElementById('sumArea').textContent = (Math.round(totalArea * 100) / 100).toLocaleString('zh-TW');
    document.getElementById('sumPrice').textContent = totalPrice.toLocaleString('zh-TW');
    summaryBox.style.display = 'flex';
  }

  function copyLineText() {
    if (items.length === 0) {
      alert('清單內目前無資料，請先新增項目。');
      return;
    }

    let totalArea = 0;
    let totalPrice = 0;

    let text = '【木地板訂製工程預估單】\n';
    text += '--------------------------------\n';
    items.forEach((item, i) => {
      totalArea += item.area;
      totalPrice += item.subtotal;
      text += `${i + 1}. 【${item.areaName}】\n`;
      text += `   規格：${item.specSummary}\n`;
      text += `   單價：$${item.unitPrice.toLocaleString('zh-TW')} / 坪\n`;
      text += `   坪數：${item.area} 坪\n`;
      text += `   小計：$${item.subtotal.toLocaleString('zh-TW')} 元\n\n`;
    });
    text += '--------------------------------\n';
    text += `合計總坪數：${Math.round(totalArea * 100) / 100} 坪\n`;
    text += `預估總金額：$${totalPrice.toLocaleString('zh-TW')} 元\n`;
    text += '※ 備註：顏色不限需打樣確認；實際金額依現場放樣與訂單規格為準。';

    navigator.clipboard.writeText(text).then(() => {
      alert('已成功複製報價文字，可直接貼至 LINE！');
    }).catch(() => {
      alert('複製失敗，請手動選取文字。');
    });
  }

  // 初始化日期與計算
  document.getElementById('todayDate').textContent = '估價日期：' + new Date().toLocaleDateString('zh-TW');
  runCalc();
</script>
